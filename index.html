/*
 * ============================================================================
 * Project: Basic Calculator Program
 * Organization: CodeAlpha Internship
 * Domain: C Programming (Task 1)
 * ============================================================================
 */

#include <stdio.h>
#include <stdlib.h>

int main(void) {
    int choice;
    double num1, num2, result;
    char continueCalc = 'y';

    do {
        printf("\n1. Addition (+)\n2. Subtraction (-)\n3. Multiplication (*)\n4. Division (/)\n5. Modulus (%%)\n6. Exit\n");
        printf("Enter your choice (1-6): ");
        scanf("%d", &choice);

        switch (choice) {
            case 1: // Addition
                printf("Enter two numbers: ");
                scanf("%lf %lf", &num1, &num2);
                printf("Result: %.2lf + %.2lf = %.2lf\n", num1, num2, num1 + num2);
                break;
            case 2: // Subtraction
                printf("Enter two numbers: ");
                scanf("%lf %lf", &num1, &num2);
                printf("Result: %.2lf - %.2lf = %.2lf\n", num1, num2, num1 - num2);
                break;
            case 3: // Multiplication
                printf("Enter two numbers: ");
                scanf("%lf %lf", &num1, &num2);
                printf("Result: %.2lf * %.2lf = %.2lf\n", num1, num2, num1 * num2);
                break;
            case 4: // Division
                printf("Enter two numbers: ");
                scanf("%lf %lf", &num1, &num2);
                if (num2 != 0)
                    printf("Result: %.2lf / %.2lf = %.2lf\n", num1, num2, num1 / num2);
                else
                    printf("Error: Division by zero is undefined!\n");
                break;
            case 5: // Modulus
                printf("Enter two integers: ");
                int a, b;
                scanf("%d %d", &a, &b);
                if (b != 0)
                    printf("Result: %d %% %d = %d\n", a, b, a % b);
                else
                    printf("Error: Modulo by zero!\n");
                break;
            case 6:
                printf("Goodbye!\n");
                return 0;
            default:
                printf("Invalid choice!\n");
        }
    } while (continueCalc == 'y' || continueCalc == 'Y');

    return 0;
}
